# 采集用到的命令（可直接复跑）
kmutil showloaded | grep -i nullmoth
system_profiler SPDisplaysDataType
nvram boot-args
sysctl debug.nvrmfb_flip_latch
plutil -p /Library/Extensions/NVRM.kext/Contents/Info.plist | grep CFBundleShortVersionString
sudo -n cat /Library/NullMoth/driver-version
ls /Library/Logs/DiagnosticReports | grep -c gpuRestart
ls /Library/Logs/DiagnosticReports/*.panic 2>/dev/null | wc -l
ps -o %cpu= -p $(pgrep -x WindowServer)   # 采样 WindowServer CPU
sudo sysctl -w debug.nvrmfb_flip_latch=2   # opt-in 高刷优化（测试后已删除持久化）
